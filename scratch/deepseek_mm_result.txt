define('MARGIN_BOTTOM_MM', 10);
define('USABLE_HEIGHT_MM', PAGE_HEIGHT_MM - MARGIN_TOP_MM - MARGIN_BOTTOM_MM);
define('PX_TO_MM', 0.264583); // 1px = 0.264583mm

// Función para convertir px a mm
function pxToMm($px) {
    return round($px * PX_TO_MM, 2);
}

// Función para obtener los bloques de cabecera
function getHeaderBlocks() {
    return ['logo', 'titulo_colegio', 'lema_colegio', 'metadatos'];
}

// Función para agrupar bloques por página
function groupBlocksByPage($blocks) {
    $pages = [];
    $currentPage = [];
    $currentY = MARGIN_TOP_MM;
    
    // Ordenar bloques por top
    usort($blocks, function($a, $b) {
        return $a['top'] <=> $b['top'];
    });
    
    $headerBlocks = getHeaderBlocks();
    $headerBlocksData = [];
    
    // Separar bloques de cabecera
    foreach ($blocks as $block) {
        if (in_array($block['id'], $headerBlocks)) {
            $headerBlocksData[] = $block;
        }
    }
    
    // Procesar bloques no-cabecera
    $nonHeaderBlocks = array_filter($blocks, function($block) use ($headerBlocks) {
        return !in_array($block['id'], $headerBlocks);
    });
    
    $currentPage = [];
    $currentY = MARGIN_TOP_MM;
    
    foreach ($nonHeaderBlocks as $block) {
        $blockTop = $block['top'];
        $blockHeight = $block['height'];
        
        // Si el bloque no cabe en la página actual
        if ($blockTop + $blockHeight > $currentY + USABLE_HEIGHT_MM) {
            // Guardar página actual
            if (!empty($currentPage)) {
                $pages[] = $currentPage;
            }
            
            // Nueva página con cabecera
            $currentPage = $headerBlocksData;
            $currentY = MARGIN_TOP_MM;
            
            // Ajustar posición del bloque
            $block['top'] = $currentY;
            $currentY += $blockHeight;
        } else {
            $currentY = $blockTop + $blockHeight;
        }
        
        $currentPage[] = $block;
    }
    
    // Agregar última página
    if (!empty($currentPage)) {
        $pages[] = $currentPage;
    }
    
    return $pages;
}

**Cambios principales implementados:**

1. **Conversión a mm en PHP**: Se define `PX_TO_MM = 0.264583` y se convierte todas las coordenadas de px a mm antes de generar el HTML.

2. **Auto-paginación en PHP**: 
   - Función `groupBlocksByPage()` que agrupa los bloques en páginas
   - Los bloques de cabecera se repiten en cada página
   - Se calcula la altura útil de página (279.4mm - 10mm - 10mm = 259.4mm)
   - Los bloques se ordenan por coordenada Y y se distribuyen en páginas

3. **Generación directa en mm**: Los divs se generan directamente con `style="left: Xmm; top: Ymm; width: Wmm;"` desde PHP, sin necesidad de JavaScript.

4. **CSS sin !important**: Se eliminaron todos los `!important` y se establecieron las dimensiones exactas en mm para `.print-container`.

5. **Estructura de páginas**: Cada página es un `.print-container` separado con `page-break-after: always` para impresión correcta.
