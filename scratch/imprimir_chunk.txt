                $html_parrafos .= '<p class="mb-2">' . $p . '</p>';
            }
        }
        return '<div' . $attrs . '><div class="block-content-texto" data-text-align="' . htmlspecialchars($align, ENT_QUOTES, 'UTF-8') . '">' . $html_parrafos . '</div></div>';
